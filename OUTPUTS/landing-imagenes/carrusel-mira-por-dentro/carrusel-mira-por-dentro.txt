<script>
(function(){
  var t=document.getElementById('mpdTrack'); if(!t) return;
  var dots=document.querySelectorAll('#mpdDots .mpd-dot'), n=dots.length, cur=0;
  function w(){return t.clientWidth;}
  function go(i){cur=Math.max(0,Math.min(n-1,i)); t.scrollTo({left:cur*w(),behavior:'smooth'});}
  document.getElementById('mpdNext').addEventListener('click',function(){go(cur>=n-1?0:cur+1);});
  document.getElementById('mpdPrev').addEventListener('click',function(){go(cur<=0?n-1:cur-1);});
  dots.forEach(function(d,i){d.addEventListener('click',function(){go(i);});});
  var tick=false;
  t.addEventListener('scroll',function(){
    if(tick) return; tick=true;
    requestAnimationFrame(function(){
      cur=Math.round(t.scrollLeft/w());
      dots.forEach(function(d,i){d.classList.toggle('on',i===cur);});
      tick=false;
    });
  },{passive:true});
  document.getElementById('mpd').addEventListener('keydown',function(e){
    if(e.key==='ArrowRight'){go(cur+1);} else if(e.key==='ArrowLeft'){go(cur-1);}
  });
})();
</script>
