<script>
/* Precios tachados de los bonos en la moneda del visitante.
   Se calculan con la proporción entre el precio que la plataforma muestra hoy ({{price}} en .now)
   y el precio base en USD (data-base en la sección, 14.99). Si no se puede leer, queda el valor en USD. */
(function(){
  function parse(t){
    var m=(t||'').match(/\d[\d.,]*/); if(!m) return null;
    var tok=m[0].replace(/[.,]+$/,''), i=Math.max(tok.lastIndexOf('.'),tok.lastIndexOf(','));
    var dec=null, tsep='', dsep='';
    if(i>-1 && tok.length-i-1<=2){dec=tok.length-i-1; dsep=tok[i];}
    var digits=tok.replace(/[.,]/g,''), num=dec===null?parseFloat(digits):parseFloat(digits.slice(0,digits.length-dec)+'.'+digits.slice(digits.length-dec));
    var others=tok.replace(/\d/g,'').replace(dsep,''); tsep=others?others[0]:'';
    return {raw:m[0],num:num,dec:dec,tsep:tsep,dsep:dsep,pre:t.slice(0,m.index),post:t.slice(m.index+m[0].length)};
  }
  function nice(v,dec){
    if(dec>0&&v<100) return Math.round(v*2)/2;               // 7.47 -> 7.5
    var p=Math.pow(10,Math.max(0,Math.floor(Math.log10(v))-2)); // 31968 -> 32000
    return Math.round(v/p)*p;
  }
  function fmt(v,p){
    var dec=(p.dec===null?0:p.dec), s=v.toFixed(dec), a=s.split('.'), int=a[0];
    if(p.tsep) int=int.replace(/\B(?=(\d{3})+(?!\d))/g,p.tsep);
    return p.pre+int+(dec?(p.dsep||'.')+a[1]:'')+p.post;
  }
  function run(){
    var sec=document.getElementById('bonos'), now=document.querySelector('.now'); if(!sec||!now) return;
    var base=parseFloat(sec.getAttribute('data-base'))||14.99, p=parse(now.textContent); if(!p||!p.num) return;
    var r=p.num/base;
    document.querySelectorAll('#bonos .bp').forEach(function(e){
      var usd=parseFloat(e.getAttribute('data-usd')), v=nice(usd*r,p.dec);
      e.textContent=fmt(v,p);
    });
  }
  if(document.readyState==='loading') document.addEventListener('DOMContentLoaded',run); else run();
  window.cecBonusPrices=run;
})();
</script>
